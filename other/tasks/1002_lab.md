```bash
stages:
  - lint
  - test
  - build
  - publish
  - deploy

variables:
  APP_VERSION: "1.0.0"
  CI_REGISTRY_IMAGE: $CI_REGISTRY_IMAGE

workflow:
  rules:
    - if: $CI_COMMIT_TAG
      when: never
    - when: always

# --- Линтинг ---
lint:
  stage: lint
  image: python:3.11-slim
  before_script:
    - pip install flake8 black
  script:
    - echo "🔍 Running linters..."
    - flake8 src/ --max-line-length=120 --statistics
    - black --check src/ tests/
    - echo "✅ Linters passed"
  rules:
    - if: $CI_PIPELINE_SOURCE == "merge_request_event"
    - if: $CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH

# --- Тестирование ---
test:
  stage: test
  image: python:3.11-slim
  before_script:
    - pip install -r requirements.txt
    - pip install pytest pytest-cov
  script:
    - echo "🧪 Running tests..."
    - pytest tests/ -v --cov=src/ --cov-report=term --cov-report=xml
  coverage: '/TOTAL.*\s+(\d+\.\d+)%/'
  artifacts:
    when: always
    reports:
      coverage_report:
        coverage_format: cobertura
        path: coverage.xml
    expire_in: 1 week
  rules:
    - if: $CI_PIPELINE_SOURCE == "merge_request_event"
    - if: $CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH

# --- Сборка образа ---
build_image:
  stage: build
  image: docker:latest
  services:
    - docker:dind
  script:
    - echo "📦 Building Docker image..."
    - docker login -u $CI_REGISTRY_USER -p $CI_REGISTRY_PASSWORD $CI_REGISTRY
    - docker build -t $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA .
    - docker tag $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA $CI_REGISTRY_IMAGE:latest
    - docker push $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA
    - docker push $CI_REGISTRY_IMAGE:latest
    - echo "✅ Image built and published"
  rules:
    - if: $CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH

# --- Деплой на staging ---
deploy_staging:
  stage: deploy
  image: alpine:latest
  before_script:
    - apk add --no-cache curl
  script:
    - echo "🚀 Deploying to Staging..."
    - echo "Image: $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA"
    - echo "Environment: staging"
    - sleep 3
    - echo "✅ Staging deployment complete"
  environment:
    name: staging
    url: https://staging.todo-api.example.com
  rules:
    - if: $CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH

  stage: deploy

  before_script:

  script:
    - echo "🚀 Deploying to Production..."
    - echo "Image: $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA"
    - echo "Version: $APP_VERSION"
    - sleep 3
    - echo "✅ Production deployment complete"
```

#### Шаг 7. Создание basic приложения

```bash
mkdir -p src tests scripts

cat > src/main.py << 'EOF'
#!/usr/bin/env python3
"""
Todo API — основное приложение.
"""

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional
import uuid
from datetime import datetime

app = FastAPI(
    title="Todo API",
    description="REST API для управления задачами",
    version="1.0.0"
)

# Модели данных
class TodoCreate(BaseModel):
    title: str
    description: Optional[str] = None
    category: str = "general"

class TodoResponse(BaseModel):
    id: str
    title: str
    description: Optional[str]
    category: str
    status: str
    created_at: datetime
    updated_at: datetime

# Хранилище (в реальном проекте — БД)
todos: dict[str, dict] = {}

@app.get("/")
async def root():
    return {"message": "Todo API v1.0.0", "docs": "/docs"}

@app.get("/health")
async def health():
    return {"status": "healthy"}

@app.get("/todos", response_model=List[TodoResponse])
async def get_todos():
    return list(todos.values())

@app.post("/todos", response_model=TodoResponse, status_code=201)
async def create_todo(todo: TodoCreate):
    todo_id = str(uuid.uuid4())
    new_todo = {
        "id": todo_id,
        **todo.model_dump(),
        "status": "active",
        "created_at": datetime.now(),
        "updated_at": datetime.now()
    }
    todos[todo_id] = new_todo
    return new_todo

@app.get("/todos/{todo_id}", response_model=TodoResponse)
async def get_todo(todo_id: str):
    if todo_id not in todos:
        raise HTTPException(status_code=404, detail="Todo not found")
    return todos[todo_id]

@app.delete("/todos/{todo_id}", status_code=204)
async def delete_todo(todo_id: str):
    if todo_id not in todos:
        raise HTTPException(status_code=404, detail="Todo not found")
    del todos[todo_id]
    return None
EOF

cat > requirements.txt << 'EOF'

import pytest
from fastapi.testclient import TestClient
from src.main import app

client = TestClient(app)

def test_root():
    response = client.get("/")
    assert response.status_code == 200
    assert response.json()["message"] == "Todo API v1.0.0"

def test_health():
    response = client.get("/health")
    assert response.status_code == 200
    assert response.json()["status"] == "healthy"

def test_create_todo():
    response = client.post("/todos", json={
        "title": "Test Task",
        "description": "Test description",
        "category": "test"
    })
    assert response.status_code == 201
    data = response.json()
    assert data["title"] == "Test Task"
    assert "id" in data

def test_get_todos():
    # Сначала создадим задачу
    client.post("/todos", json={"title": "Task 1"})
    
    response = client.get("/todos")
    assert response.status_code == 200
    data = response.json()
    assert len(data) >= 1

def test_get_todo_not_found():
    response = client.get("/todos/non-existent-id")
    assert response.status_code == 404
EOF
```
